cmake_minimum_required(VERSION 3.16)
project(piper_sim_devices)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
set(NANKAI_MAIN_SOURCE "" CACHE PATH "External NankaiRoboCup2026 checkout (read only)")
if(NANKAI_MAIN_SOURCE)
  find_package(Threads REQUIRED)
  find_package(Eigen3 REQUIRED)
  foreach(dep rclcpp rclcpp_action rcl_interfaces sensor_msgs control_msgs diagnostic_msgs nankai_robocup_interfaces)
    find_package(${dep} REQUIRED)
  endforeach()
  set(core "${NANKAI_MAIN_SOURCE}/src/manipulation/nankai_robocup_arm_control")
  # Upstream does not export arm_core. Compile its current files in our build
  # directory; never edit the checkout or vendor a second copy.
  add_library(nankai_control_core STATIC ${core}/src/protocol.cpp ${core}/src/control.cpp
    ${core}/src/kinematics.cpp ${core}/src/motion.cpp ${core}/src/transport.cpp)
  target_include_directories(nankai_control_core PUBLIC ${core}/include)
  target_link_libraries(nankai_control_core PUBLIC Threads::Threads Eigen3::Eigen)
  add_executable(isaac_arm_driver src/arm_driver.cpp)
  target_include_directories(isaac_arm_driver PRIVATE include)
  target_link_libraries(isaac_arm_driver nankai_control_core)
  target_compile_options(isaac_arm_driver PRIVATE -Wall -Wextra -Wpedantic)
  ament_target_dependencies(isaac_arm_driver rclcpp rclcpp_action rcl_interfaces sensor_msgs control_msgs diagnostic_msgs nankai_robocup_interfaces)
  install(TARGETS isaac_arm_driver DESTINATION lib/${PROJECT_NAME})
  if(BUILD_TESTING)
    find_package(ament_cmake_test REQUIRED)
    add_executable(controller_check test/controller_check.cpp)
    target_include_directories(controller_check PRIVATE include)
    target_link_libraries(controller_check nankai_control_core)
    ament_add_test(controller_check COMMAND $<TARGET_FILE:controller_check> GENERATE_RESULT_FOR_RETURN_CODE_ZERO)
    foreach(test gripper_check motion_check)
      add_executable(upstream_${test} ${core}/test/${test}.cpp)
      target_link_libraries(upstream_${test} nankai_control_core)
      ament_add_test(upstream_${test} COMMAND $<TARGET_FILE:upstream_${test}> GENERATE_RESULT_FOR_RETURN_CODE_ZERO)
    endforeach()
  endif()
endif()
install(PROGRAMS scripts/joint_state_relay DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch DESTINATION share/${PROJECT_NAME} PATTERN "__pycache__" EXCLUDE)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_package()
